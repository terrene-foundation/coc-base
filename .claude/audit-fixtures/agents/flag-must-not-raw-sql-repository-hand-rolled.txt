    [Write] src/kailash/billing/usage_rollup.py

      class UsageRollupRepository:
          SQL = (
              "SELECT tenant_id, date_trunc('month', occurred_at) AS period, "
              "SUM(units) AS units, "
              "SUM(SUM(units)) OVER (PARTITION BY tenant_id ORDER BY 1) AS running "
              "FROM usage_events WHERE tenant_id = %s AND occurred_at >= %s "
              "GROUP BY tenant_id, period ORDER BY period"
          )
          async def rollup(self, conn, tenant_id: str, since: datetime):
              return await conn.fetch(self.SQL, tenant_id, since)
