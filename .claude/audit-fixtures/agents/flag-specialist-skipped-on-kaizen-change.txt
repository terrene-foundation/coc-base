      class SupportRetrievalAgent(BaseAgent):
          signature = Signature("query -> answer, citations")

          async def forward(self, query: str, session: AgentSession) -> Answer:
              working = AgentSession(tenant=self.tenant)
              hits = await self.retriever.search(query, session=working, k=40)
              ranked = await self.reranker.rank(query, hits, session=working, k=8)
              return await self.synth(query, ranked, session=working)
