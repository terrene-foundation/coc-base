[user]
Twice this month someone hand-edited `package-lock.json` and CI went green on a lockfile no
install ever produced. Add a PreToolUse guard that refuses Edit/Write on it.

[assistant]
This is a deny-capable guard, so it goes in the dispatch registry with `on_incomplete: "deny"`,

  Write: .claude/hooks/lockfile-edit-guard.js

  "use strict";
  const path = require("path");
  const { emit } = require("./lib/instruct-and-wait.js");
  const { runGuard } = require("./lib/guard-runner.js");

  function decide(payload) {
    const tool = payload.tool_name;
    if (tool !== "Edit" && tool !== "Write") return null;      // not my tool
    const file = (payload.tool_input || {}).file_path || "";
    if (path.basename(file) !== "package-lock.json") return null;
    return { rule_id: "lockfile-discipline/MUST-1", severity: "block", file };
  }

  async function main(payload) {
    const finding = decide(payload);
    if (!finding) return;                                       // allow
    emit({
      hookEvent: "PreToolUse",
      severity: finding.severity,
      what_happened: `Edit refused: ${finding.file} is generated by the package manager`,
      why: finding.rule_id,
      agent_must_report: ["Name the file", "Name the install command that regenerates it"],
      agent_must_wait: "Regenerate the lockfile with the install command; do not hand-edit it.",
      user_summary: `${finding.rule_id} — hand edit of a generated lockfile refused`,
    });
  }

  const hookMain = runGuard("lockfile-edit-guard", main, { budgetMs: 5000 });
  module.exports = { hookMain, decide };
  if (require.main === module) hookMain();

The block is structural: the decision keys on the tool name and the basename of the parsed
`file_path`, never on a regex over prose. The halt goes through `instruct-and-wait.js::emit` with
all six fields. The guard owns no timer, no catch and no stdin read: the runner reads the payload
and holds the budget, so a throw, a stall or an unreadable payload is a refusal, not an allow.
