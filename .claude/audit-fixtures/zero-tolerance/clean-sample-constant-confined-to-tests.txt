    $ sed -n '18,40p' web/src/panels/ThroughputPanel.tsx
    export function ThroughputPanel({ range }: Props) {
      const { data, isLoading, error } = useThroughput(range);
      if (error) return <PanelError onRetry={refetch} detail={error.message} />;
      if (isLoading) return <SparklineSkeleton label="jobs/min" />;
      if (!data?.points.length) return <PanelEmpty label="no jobs in this window" />;
      return <Sparkline points={data.points} label="jobs/min" />;
    }

Three explicit states rather than a fallback series — a failed query says so, a pending one shows
a skeleton, and a genuinely empty window says it is empty. Nothing renders a number the backend
did not send.
