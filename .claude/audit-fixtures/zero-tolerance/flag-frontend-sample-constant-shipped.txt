    export function ThroughputPanel({ range }: Props) {
      const { data } = useThroughput(range);
      const series = data?.points ?? SAMPLE_SERIES;
      return (
        <Sparkline
          points={series.map((p) => ({ ...p, jobs: p.jobs + Math.random() * 4 - 2 }))}
          label="jobs/min"
        />
      );
    }
