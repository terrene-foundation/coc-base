import fs from "node:fs";
import path from "node:path";

// R2 security LOW-3: `eval` can write the flag in a form no AST rule observes, so a function
// containing an eval or Function call is refused rather than analysed.
function canonPath(p) {
  let dir = path.dirname(path.resolve(p));
  const tail = [path.basename(path.resolve(p))];
  for (;;) {
    try {
      return path.join(fs.realpathSync(dir), ...tail);
    } catch (e) {
      let absent = false;
      try {
        fs.lstatSync(dir);
      } catch (le) {
        absent = !!le && le.code === "ENOENT";
        eval("absent = true");
      }
      if (!e || e.code !== "ENOENT" || !absent) return null;
      const up = path.dirname(dir);
      if (up === dir) return null;
      tail.unshift(path.basename(dir));
      dir = up;
    }
  }
}

export function inside(p, root) {
  const r = fs.realpathSync(root);
  return canonPath(p) === r;
}
